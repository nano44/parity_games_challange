import PGLemmas.Spec

namespace PG.Spec

/-- G: every S1 call on a well-formed game satisfies the key rule. -/
def KeyRuleStatement : Prop :=
  ∀ h : SGame, h.WF → ∀ s : Array ℕ, h.StratWF 0 s →
    (Eng.ofGame h).keyRule s ((Eng.ofGame h).evaluate s) = true

/-- C: S4a's call satisfies the key rule: in dual(H) with the dual-Even strategy τ fixed, every chooser vertex (an Even
vertex of g) that is not its own cycle node takes the tied successor that R24's rule names. -/
def KeyRuleS4aStatement : Prop :=
  ∀ g : SGame, g.WF → ∀ sigma tau : Array ℕ, g.StratWF 0 sigma → g.StratWF 1 tau →
    ∀ Is : List (ℕ × ℕ), (∀ p ∈ Is, p.1 < g.n ∧ g.owner[p.1]! = 0 ∧ p.2 ∈ g.succ[p.1]!) →
      (Eng.ofGame (arenaH g sigma tau Is).dual).keyRule tau
        ((Eng.ofGame (arenaH g sigma tau Is).dual).evaluate tau) = true

end PG.Spec
